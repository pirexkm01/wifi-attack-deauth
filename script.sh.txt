#!/bin/bash

clear

# =========================
# COLORS
# =========================
GREEN="\e[1;32m"
WHITE="\e[1;37m"
RESET="\e[0m"

# =========================
# HEADER (KAMAL ADDED)
# =========================
echo -e "${GREEN}"
echo "======================================================"
echo " ██╗  ██╗ █████╗ ███╗   ███╗ █████╗ ██╗     "
echo " ██║ ██╔╝██╔══██╗████╗ ████║██╔══██╗██║     "
echo " █████╔╝ ███████║██╔████╔██║███████║██║     "
echo " ██╔═██╗ ██╔══██║██║╚██╔╝██║██╔══██║██║     "
echo " ██║  ██╗██║  ██║██║ ╚═╝ ██║██║  ██║███████╗"
echo " ╚═╝  ╚═╝╚═╝  ╚═╝╚═╝     ╚═╝╚═╝  ╚═╝╚══════╝"
echo "======================================================"
echo -e "${WHITE}"
echo "              INSTAGRAM : @SIIR_PIREX"
echo "======================================================"
echo -e "${RESET}"

sleep 1

# =========================
# INTERFACES
# =========================
echo -e "${GREEN}[+] Available interfaces:${RESET}"
iwconfig 2>/dev/null | grep "IEEE"

echo
read -p "Enter interface (example wlan0): " iface

# =========================
# MONITOR MODE
# =========================
echo
echo -e "${GREEN}[+] Stopping services...${RESET}"
sudo airmon-ng check kill

echo
echo -e "${GREEN}[+] Starting monitor mode...${RESET}"
sudo airmon-ng start "$iface"

echo
read -p "Enter monitor interface (example wlan0mon): " mon

# =========================
# OUTPUT PATH
# =========================
mkdir -p "$HOME/Desktop"
output="$HOME/Desktop/hack1"

clear

# =========================
# TERMINAL DETECTION
# =========================
TERM_CMD=""

if command -v x-terminal-emulator >/dev/null 2>&1; then
    TERM_CMD="x-terminal-emulator"
elif command -v gnome-terminal >/dev/null 2>&1; then
    TERM_CMD="gnome-terminal"
elif command -v xfce4-terminal >/dev/null 2>&1; then
    TERM_CMD="xfce4-terminal"
elif command -v konsole >/dev/null 2>&1; then
    TERM_CMD="konsole"
else
    echo "[!] No graphical terminal found!"
    exit 1
fi

echo -e "${GREEN}[+] Using terminal:${RESET} $TERM_CMD"

# =========================
# SCAN WINDOW
# =========================
echo
echo -e "${GREEN}[+] Opening scan window...${RESET}"

$TERM_CMD -e bash -c "echo -e '\e[1;32mSCAN MODE ACTIVE\e[0m'; sudo airodump-ng $mon; exec bash" &

sleep 2

echo
read -p "Press ENTER after selecting target..."

# =========================
# TARGET INPUT
# =========================
read -p "Enter BSSID: " bssid
read -p "Enter Channel: " channel

# =========================
# CAPTURE WINDOW
# =========================
echo
echo -e "${GREEN}[+] Opening capture window...${RESET}"

$TERM_CMD -e bash -c "echo -e '\e[1;32mCAPTURE MODE ACTIVE\e[0m'; sudo airodump-ng $mon -w $output -c $channel --bssid $bssid; exec bash" &

echo
echo -e "${GREEN}[+] DONE ✔${RESET}"
echo -e "${WHITE}[+] Saved in: $output${RESET}"
echo -e "${GREEN}[+] Instagram: @SIIR_PIREX (always visible)${RESET}"
